cmake_minimum_required(VERSION 3.24)
project(qwen_resident_native LANGUAGES C CXX)

if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
  message(FATAL_ERROR "Run tools/build-qwen.ps1 to obtain the pinned llama.cpp source")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Link the backend into our C ABI library" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "No external backend discovery" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "Use upstream tokenizer and Jinja chat handling" FORCE)
foreach(option LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL LLAMA_SUBPROCESS LLAMA_LLGUIDANCE)
  set(${option} OFF CACHE BOOL "Not required by this embedded backend" FORCE)
endforeach()
set(LLAMA_BUILD_COMMIT "b54752df40e4d8e53d34ef5a598336b4e4871d39")
add_subdirectory(${LLAMA_SOURCE_DIR} llama EXCLUDE_FROM_ALL)
add_library(qwen_native SHARED qwen_native.cpp)
target_compile_features(qwen_native PRIVATE cxx_std_17)
target_link_libraries(qwen_native PRIVATE llama llama-common)
target_compile_definitions(qwen_native PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(qwen_native PRIVATE /W4 /EHsc /utf-8 /Z7)
  target_link_options(qwen_native PRIVATE /DEBUG /MAP)
endif()
enable_testing()
add_executable(qwen_stream_test stream_test.cpp)
target_compile_features(qwen_stream_test PRIVATE cxx_std_17)
add_test(NAME qwen_stream COMMAND qwen_stream_test)
